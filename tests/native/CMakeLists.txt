cmake_minimum_required(VERSION 3.16)
project(eez_generated_lvgl_tests C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 11)
get_filename_component(STUDIO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LVGL_VERSION "8.4.0" CACHE STRING "Pinned LVGL version")
set(LVGL_SOURCE_DIR "" CACHE PATH "Pinned LVGL source checkout")
set(LVGL_CONFIG_FILE "" CACHE FILEPATH "Matching Studio runtime lv_conf.h")
set(GENERATED_DIR "${STUDIO}/build/lvgl-regression/${LVGL_VERSION}")
file(READ "${LVGL_CONFIG_FILE}" CONFIG)
string(REGEX REPLACE "#define LV_USE_FREETYPE +1" "#define LV_USE_FREETYPE 0" CONFIG "${CONFIG}")
string(REGEX REPLACE "#define LV_TICK_CUSTOM +1" "#define LV_TICK_CUSTOM 0" CONFIG "${CONFIG}")
string(REPLACE "extern void emscripten_force_exit(int); emscripten_force_exit(1);" "abort();" CONFIG "${CONFIG}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/lv_conf.h"
    "#ifndef __ASSEMBLER__\n#include <stdlib.h>\n#endif\n${CONFIG}")
set(LV_CONF_PATH "${CMAKE_CURRENT_BINARY_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_BINARY_DIR}/lv_conf.h" CACHE PATH "" FORCE)
get_filename_component(LVGL_PARENT "${LVGL_SOURCE_DIR}" DIRECTORY)
include_directories("${CMAKE_CURRENT_BINARY_DIR}" "${LVGL_SOURCE_DIR}" "${LVGL_PARENT}")
add_compile_definitions(LV_CONF_INCLUDE_SIMPLE LV_LVGL_H_INCLUDE_SIMPLE)
if(LVGL_VERSION VERSION_LESS "9.3.0")
    add_compile_definitions(LV_CONF_PATH=${CMAKE_CURRENT_BINARY_DIR}/lv_conf.h)
endif()
add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g -O1)
add_link_options(-fsanitize=address,undefined)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_SOURCE_DIR}" lvgl)
foreach(MODE IN ITEMS flow no-flow)
    add_executable(generated-${MODE} main.cpp "${GENERATED_DIR}/${MODE}/screens.c")
    target_include_directories(generated-${MODE} PRIVATE "${GENERATED_DIR}/${MODE}")
    target_link_libraries(generated-${MODE} lvgl m)
    target_compile_options(generated-${MODE} PRIVATE "$<$<COMPILE_LANGUAGE:C>:-Werror=implicit-function-declaration>")
endforeach()
set(FRAMEWORK "${STUDIO}/resources/eez-framework-amalgamation")
target_sources(generated-flow PRIVATE "${GENERATED_DIR}/flow/assets.c"
    "${FRAMEWORK}/eez-flow.cpp" "${FRAMEWORK}/eez-flow-lz4.c" "${FRAMEWORK}/eez-flow-sha256.c")
target_compile_definitions(generated-flow PRIVATE EEZ_FOR_LVGL TEST_FLOW)
target_include_directories(generated-flow PRIVATE "${FRAMEWORK}")
enable_testing()
add_test(NAME generated-flow COMMAND generated-flow)
add_test(NAME generated-no-flow COMMAND generated-no-flow)
